# 前端项目结构说明（更新版）

本项目是一个 **Vue 3 + Pinia + Vue Router** 的研究型可视化前端，用于 **故事演化 / 语义流动 / 群体与遗传实验** 的交互与分析。整体设计强调：

- **首页独立**
- **统一壳布局（AppShell）**
- **按研究主题分区路由**
- **可复用组件 + 清晰数据流**

------

## 根目录

- **vue.config.js**
   本地开发代理配置：
  - `/api` → `http://127.0.0.1:8000`
  - `/ws` → WebSocket 转发
     前端代码中无需写后端真实地址，支持 REST + 实时流式加点。

------

## src 入口层

- **main.js**
   Vue 应用入口：

  - 创建 app
  - 挂载 Pinia（状态管理）
  - 挂载 Router


  ```js
  export function openStoryWS({ storyId, onMessage, onOpen, onClose, onError }) {
    // 开发阶段直接连后端，避免 WebSocket proxy 问题
    const ws = new WebSocket(`ws://localhost:8000/ws/story/${storyId}`)
  
    ws.onopen = () => {
      if (onOpen) onOpen()
    }
  
    ws.onclose = () => {
      if (onClose) onClose()
    }
  
    ws.onerror = (e) => {
      if (onError) onError(e)
    }
  
    ws.onmessage = (evt) => {
      try {
        const data = JSON.parse(evt.data)
        if (onMessage) onMessage(data)
      } catch (err) {
        // 避免 ESLint no-empty 报错，同时保留调试信息
        console.warn('[WS] Received non-JSON message:', evt.data)
      }
    }
  
    return ws
  }
  
  ```

## src/store（状态管理）

- **store/storyStore.js**
   Pinia Store，统一管理：

  - `storyId`
  - `turns`
  - loading / error
  - 回放状态（进度 / 速度）

  同时提供图表用 getters：

  - `labels`
  - `flowSeries`
  - `entropySeries`
  
  ```js
  import { defineStore } from 'pinia'
  import { createStory, continueStory, getStory } from '../api/story'
  
  function clamp01(x) {
    if (Number.isNaN(x)) return 0
    return Math.max(0, Math.min(1, x))
  }
  
  export const useStoryStore = defineStore('story', {
    state: () => ({
      storyId: localStorage.getItem('story_id') || null,
      turns: JSON.parse(localStorage.getItem('story_turns') || '[]'),
      loading: false,
      error: null,
  
      // replay
      replayIndex: 0,
      replayPlaying: false,
      replaySpeedMs: 450,
  
      // visualization
      staticData: null,
    }),
  
    getters: {
      flowSeries(state) {
        return state.turns.map((t) => clamp01(Number(t.flow_score ?? 0)))
      },
      entropySeries(state) {
        return state.turns.map((t) => clamp01(Number(t.entropy_score ?? 0)))
      },
      labels(state) {
        return state.turns.map((t) => `T${t.turn}`)
      },
  
      latestEntropy(state) {
        const last = state.turns[state.turns.length - 1]
        return clamp01(Number(last?.entropy_score ?? 0))
      },
  
      // ✅ 新增：最新 entropy 相比上一条的变化量（默认按“百分点 pp”返回）
      latestEntropyDelta(state) {
        const n = state.turns.length
        if (n < 2) return null
  
        const last = clamp01(Number(state.turns[n - 1]?.entropy_score ?? 0))
        const prev = clamp01(Number(state.turns[n - 2]?.entropy_score ?? 0))
  
        // 1) 按百分点显示（推荐，与仪表盘百分比口径一致）
        return (last - prev) * 100
  
        // 2) 如果你想显示 0~1 量纲（比如 +0.03 / -0.03），用这一行替换上面 return：
        // return last - prev
      },
  
      visibleTurns(state) {
        if (!state.replayPlaying && state.replayIndex === 0) return state.turns
        return state.turns.slice(0, Math.max(0, state.replayIndex))
      },
    },
  
    actions: {
      saveState() {
        if (this.storyId) localStorage.setItem('story_id', this.storyId)
        localStorage.setItem('story_turns', JSON.stringify(this.turns))
      },
  
      async ensureStory() {
        if (this.storyId) return this.storyId
        const res = await createStory()
        this.storyId = res.story_id
        this.turns = []
        this.error = null
        this.saveState()
        return this.storyId
      },
  
      async loadStory(storyId) {
        this.loading = true
        this.error = null
        try {
          const res = await getStory(storyId)
          this.storyId = res.story_id
          this.turns = Array.isArray(res.turns) ? res.turns : []
          this.saveState()
        } catch (e) {
          this.error = e?.message || 'Load story failed'
        } finally {
          this.loading = false
        }
      },
  
      async restContinue({ userText, rounds = 1, mode = 'human_ai' }) {
        this.loading = true
        this.error = null
        try {
          const sid = await this.ensureStory()
          const res = await continueStory({
            story_id: sid,
            user_text: userText,
            rounds,
            mode,
          })
          const newTurns = Array.isArray(res.new_turns) ? res.new_turns : []
          this.turns = this.turns.concat(newTurns)
          this.saveState()
        } catch (e) {
          this.error = e?.message || 'Continue failed'
        } finally {
          this.loading = false
        }
      },
  
      pushTurnFromWS(item) {
        this.turns = this.turns.concat([item])
        this.saveState()
      },
  
      resetReplay() {
        this.replayIndex = 0
        this.replayPlaying = false
      },
  
      async playReplay() {
        this.replayPlaying = true
        this.replayIndex = 0
        while (this.replayPlaying && this.replayIndex < this.turns.length) {
          await new Promise((r) => setTimeout(r, this.replaySpeedMs))
          this.replayIndex += 1
        }
        this.replayPlaying = false
      },
  
      pauseReplay() {
        this.replayPlaying = false
      },
  
      async fetchStaticData() {
        if (this.staticData && Object.keys(this.staticData).length > 0) return
        try {
          const { getStaticData } = await import('../api/story')
          const data = await getStaticData()
          this.staticData = data
        } catch (e) {
          console.warn('Fetch static data failed:', e)
        }
      },
    },
  })
  
  ```

- **MessageList.vue(没有粘贴样式部分)**
   消息渲染组件：
  
  - Human / AI 气泡
  - turn 编号
  - Flow / Entropy 分数
  - 自动滚动
  
  ```vue
  <template>
    <div class="list" ref="listEl">
      <div
        v-for="(m, idx) in visibleMessages"
        :key="msgKey(m, idx)"
        class="msg"
        :class="m.author === 'human' ? 'human' : 'ai'"
      >
        <div class="meta">
          <span class="tag">{{ m.author === 'human' ? 'Human' : 'AI' }}</span>
          <span class="turn">T{{ m.turn }}</span>
          <span class="scores">
            flow {{ fmt(m.flow_score) }} · entropy {{ fmt(m.entropy_score) }}
          </span>
        </div>
  
        <div class="bubble">
          <TypingEffect
            v-if="m.author === 'ai'"
            :text="m.text"
            :speed="30"
            @done="onTypeDone"
          />
          <span v-else>{{ m.text }}</span>
        </div>
      </div>
  
      <div v-if="!messages || !messages.length" class="empty">
        No messages yet.
      </div>
    </div>
  </template>
  
  <script>
  import TypingEffect from './TypingEffect.vue'
  
  export default {
    name: 'MessageList',
    components: { TypingEffect },
    props: {
      messages: { type: Array, default: () => [] },
    },
    data() {
      return {
        visibleCount: 0,
        advanceTimer: null,
      }
    },
    computed: {
      visibleMessages() {
        return this.messages.slice(0, this.visibleCount)
      },
    },
    watch: {
      messages: {
        immediate: true,
        handler(newVal, oldVal) {
          const newArr = Array.isArray(newVal) ? newVal : []
          const oldArr = Array.isArray(oldVal) ? oldVal : []
  
          const newLen = newArr.length
          const oldLen = oldArr.length
  
          // 清空
          if (newLen === 0) {
            this.visibleCount = 0
            this.clearAdvanceTimer()
            return
          }
  
          // ✅ 只在“真正重载/换故事”时重置：
          // 1) 变短（通常是 resetReplay / clear / loadStory 的重置过程）
          const shrunk = newLen < oldLen
  
          // 2) 第一条不一样（换 story / reload 替换了整批 turns）
          const firstChanged =
            oldLen > 0 &&
            (newArr[0]?.turn !== oldArr[0]?.turn ||
              newArr[0]?.author !== oldArr[0]?.author ||
              newArr[0]?.text !== oldArr[0]?.text)
  
          if (shrunk || firstChanged) {
            this.visibleCount = 1 // 直接从第一条开始展示
            this.clearAdvanceTimer()
            // 如果第一条是 human，自动推进下一条（模拟之前逻辑）
            this.$nextTick(() => this.autoAdvanceIfHuman())
            return
          }
  
          // ✅ 初次进入：至少展示第一条
          if (oldLen === 0 && this.visibleCount === 0 && newLen > 0) {
            this.visibleCount = 1
            this.$nextTick(() => this.autoAdvanceIfHuman())
            return
          }
  
          // ✅ 正常追加：如果刚好显示到了旧末尾，推进显示下一条
          if (newLen > oldLen && this.visibleCount === oldLen) {
            this.visibleCount = oldLen + 1
            this.$nextTick(() => this.autoAdvanceIfHuman())
          }
        },
      },
  
      visibleCount() {
        // 每次显示增加，都滚到底
        this.scrollToBottom()
      },
    },
    methods: {
      msgKey(m, idx) {
        // 稳定 key：优先 _uid/id，其次 author+turn，最后兜底 idx
        return m?._uid || m?.id || `${m?.author || 'u'}-${m?.turn ?? 'x'}-${idx}`
      },
  
      fmt(x) {
        const n = Number(x ?? 0)
        return Number.isFinite(n) ? n.toFixed(2) : '0.00'
      },
  
      clearAdvanceTimer() {
        if (this.advanceTimer) {
          clearTimeout(this.advanceTimer)
          this.advanceTimer = null
        }
      },
  
      autoAdvanceIfHuman() {
        if (this.visibleCount <= 0 || this.visibleCount > this.messages.length) return
        const lastMsg = this.messages[this.visibleCount - 1]
        if (lastMsg && lastMsg.author !== 'ai') {
          // human 没有 done 事件，自动推进
          this.onTypeDone()
        }
      },
  
      onTypeDone() {
        this.scrollToBottom()
  
        if (this.visibleCount < this.messages.length) {
          this.clearAdvanceTimer()
          this.advanceTimer = setTimeout(() => {
            this.visibleCount++
            this.advanceTimer = null
            // 如果下一条还是 human，继续推进
            this.$nextTick(() => this.autoAdvanceIfHuman())
          }, 500)
        }
      },
  
      scrollToBottom() {
        this.$nextTick(() => {
          const el = this.$refs.listEl
          if (el) el.scrollTop = el.scrollHeight
        })
      },
    },
    beforeUnmount() {
      this.clearAdvanceTimer()
    },
  }
  </script>
  ```
  
  
  
- **TypingEffect.vue**

  ```vue
  <template>
    <span>{{ displayedText }}</span>
  </template>
  
  <script>
  export default {
    name: 'TypingEffect',
    props: {
      text: { type: String, required: true },
      speed: { type: Number, default: 30 }, // ms per char
    },
    data() {
      return {
        displayedText: '',
        timer: null,
      }
    },
    watch: {
      text: {
        immediate: true,
        handler(val) {
          if (!val) {
            this.displayedText = ''
            return
          }
          // 如果已经是完整显示的，就不重置动画了（避免滚动时反复打字）
          // 只有当新文本比当前显示的长，且是追加的时候才动画
          // 简单起见：每次 text 变动，如果是增量变动，就从当前位置继续打
          if (val.startsWith(this.displayedText) && this.displayedText.length > 0) {
             this.typeNext()
          } else {
             // 全新的文本，重置
             this.displayedText = ''
             this.typeNext()
          }
        },
      },
    },
    beforeUnmount() {
      clearTimeout(this.timer)
    },
    methods: {
      typeNext() {
        clearTimeout(this.timer)
        if (this.displayedText.length < this.text.length) {
          // 计算下一个要显示的字符索引
          const nextChar = this.text[this.displayedText.length]
          this.displayedText += nextChar
          
          // 随机速度增加真实感
          const randomSpeed = this.speed + (Math.random() - 0.5) * 20
          this.timer = setTimeout(this.typeNext, Math.max(10, randomSpeed))
        } else {
          // 完成
          this.$emit('done')
        }
      },
    },
  }
  </script>
  
  ```
